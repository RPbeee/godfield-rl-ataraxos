cmake_minimum_required(VERSION 3.28)
project(gfrl LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(GFRL_SANITIZE "Instrument C++ core, engine and tests with ASan/UBSan" OFF)
set(BUILD_TESTING OFF CACHE BOOL "Disable upstream tests" FORCE)
set(GF_SANITIZE ${GFRL_SANITIZE} CACHE BOOL "Instrument upstream engine" FORCE)
add_subdirectory(third_party/goodfield EXCLUDE_FROM_ALL)
find_package(Threads REQUIRED)
add_library(gfrl_core STATIC cpp/gfrl_core/bots.cpp cpp/gfrl_core/selection.cpp cpp/gfrl_core/vec_env.cpp cpp/gfrl_core/encoder.cpp cpp/gfrl_core/history.cpp)
set_target_properties(gfrl_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(gfrl_core PUBLIC cpp)
target_link_libraries(gfrl_core PUBLIC godfield Threads::Threads)
target_compile_options(gfrl_core PRIVATE -Wall -Wextra -Wpedantic)
if(GFRL_SANITIZE)
  target_compile_options(gfrl_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(gfrl_core PUBLIC -fsanitize=address,undefined)
endif()
# Keep the Release Python module intact when building instrumented C++ targets.
if(NOT GFRL_SANITIZE)
  find_package(Python 3.12 REQUIRED COMPONENTS Interpreter Development.Module)
  execute_process(COMMAND "${Python_EXECUTABLE}" -m nanobind --cmake_dir
    OUTPUT_VARIABLE nanobind_DIR OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  find_package(nanobind CONFIG REQUIRED)
  nanobind_add_module(_core cpp/gfrl_core/bindings.cpp)
  target_link_libraries(_core PRIVATE gfrl_core)
  set_target_properties(_core PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/python/gfrl")
endif()
add_executable(gfrl_bench cpp/gfrl_core/bench.cpp)
target_link_libraries(gfrl_bench PRIVATE gfrl_core)
enable_testing()
add_executable(gfrl_tests tests/cpp/test_core.cpp)
target_link_libraries(gfrl_tests PRIVATE gfrl_core)
foreach(case tree legality determinism automatic invalid allocation)
  add_test(NAME gfrl_${case} COMMAND gfrl_tests ${case})
endforeach()
add_executable(gfrl_observation_tests tests/cpp/test_observation.cpp)
target_link_libraries(gfrl_observation_tests PRIVATE gfrl_core)
foreach(case leak slots history ring)
  add_test(NAME gfrl_${case} COMMAND gfrl_observation_tests ${case})
endforeach()
add_executable(gfrl_bot_tests tests/cpp/test_bots.cpp)
target_link_libraries(gfrl_bot_tests PRIVATE gfrl_core)
foreach(case leaks pairs determinism legal greedy external)
  add_test(NAME gfrl_bots_${case} COMMAND gfrl_bot_tests ${case})
endforeach()
add_executable(gfrl_search_tests tests/cpp/test_search.cpp)
target_link_libraries(gfrl_search_tests PRIVATE gfrl_core)
add_test(NAME gfrl_search COMMAND gfrl_search_tests)
